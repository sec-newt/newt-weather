# Every Android permission the installed app may request, one per line.
# The build fails if the APK asks for anything not listed here, so a new
# permission (for example location, or seeing every installed app through
# QUERY_ALL_PACKAGES) can never sneak in unnoticed.
#
# The forecast from the National Weather Service (api.weather.gov), the only
# place the app connects to. Newt Weather is the one Newt app with internet,
# so the others can show the weather while staying offline.
android.permission.INTERNET
#
# A notification for severe-weather Warnings only; asked for when you tap
# "Turn on warnings".
android.permission.POST_NOTIFICATIONS
#
# Follows Newt Home's look (terminal or clean). Signature-level, so Newt Home
# grants it only to Newt-signed apps; it shares nothing but which look.
io.github.secnewt.home.permission.READ_LOOK
#
# Added by WorkManager, which runs the background refresh:
# - only refresh when the phone is online
android.permission.ACCESS_NETWORK_STATE
# - keep the phone awake for the few seconds a refresh takes
android.permission.WAKE_LOCK
# - put the refresh schedule back after the phone restarts
android.permission.RECEIVE_BOOT_COMPLETED
#
# Added automatically by AndroidX so the app's own broadcast receivers stay private:
io.github.secnewt.weather.DYNAMIC_RECEIVER_NOT_EXPORTED_PERMISSION
io.github.secnewt.weather.debug.DYNAMIC_RECEIVER_NOT_EXPORTED_PERMISSION
